// 写法一：async / await（本页采用） async function xhsCook() { // 谁先完成用谁（线上 6s / 线下 8s） const xhsMat = await Promise.race([ xhsBuyOnline(), xhsBuyOffline() ]); await xhsPickUp(xhsMat); // 取原料 2s // 和面、和馅同时做，两个都完成 await Promise.all([ xhsKneadDough(), xhsMixFilling() ]); await xhsWrap(); // 包饺子 5s await xhsBoil(); // 煮饺子 4s await xhsEat(); // 吃饺子 2s }
// 写法二：Promise 链（等价写法） xhsBuyOnlineOrOffline() { return Promise.race([ xhsBuyOnline(), xhsBuyOffline() ]); } xhsBuyOnlineOrOffline() .then(xhsMat => xhsPickUp(xhsMat)) .then(() => Promise.all([ xhsKneadDough(), xhsMixFilling() ])) .then(() => xhsWrap()) .then(() => xhsBoil()) .then(() => xhsEat()) .catch(xhsErr => console.error(xhsErr));
